static u8 GetCurrentRegionCardType(void)
{
    if (IsPlayerInKantoRegion())

        return CARD_TYPE_FRLG;  // Kanto style
    }
    else if (IsPlayerInJohtoRegion())
    {
        return CARD_TYPE_FRLG;  // Johto style (using Kanto card style for now)
    }

}
